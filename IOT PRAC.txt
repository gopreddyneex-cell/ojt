#gps.py#

import time
import serial
import pynmea2

# Serial port connected to the GPS module
port = "/dev/ttyAMA0"

# Create serial object
ser = serial.Serial(
    port,
    baudrate=9600,
    timeout=0.5
)

print("Reading GPS data...")

try:
    while True:
        data = ser.readline().decode(
            "ascii", errors="ignore"
        ).strip()

        if data.startswith("$GPGGA"):
            try:
                msg = pynmea2.parse(data)

                print(msg)
                print("Latitude:", msg.latitude)
                print("Longitude:", msg.longitude)

            except pynmea2.ParseError:
                print("Invalid GPS data")

        time.sleep(2)

except KeyboardInterrupt:
    print("GPS program stopped")

finally:
    ser.close()

#lcd_i2c.py#

import smbus
import time

# Device parameters
I2C_ADDR = 0x27
LCD_WIDTH = 16

# LCD commands and modes
LCD_CHR = 1
LCD_CMD = 0

LCD_LINE_1 = 0x80
LCD_LINE_2 = 0xC0
LCD_LINE_3 = 0x94
LCD_LINE_4 = 0xD4

LCD_BACKLIGHT = 0x08
ENABLE = 0b00000100

# Timing constants
E_PULSE = 0.0005
E_DELAY = 0.0005

# Open I2C bus
bus = smbus.SMBus(1)


def lcd_init():
    lcd_byte(0x33, LCD_CMD)
    lcd_byte(0x32, LCD_CMD)
    lcd_byte(0x06, LCD_CMD)
    lcd_byte(0x0C, LCD_CMD)
    lcd_byte(0x28, LCD_CMD)
    lcd_byte(0x01, LCD_CMD)
    time.sleep(E_DELAY)


def lcd_byte(bits, mode):
    bits_high = mode | (bits & 0xF0) | LCD_BACKLIGHT
    bits_low = mode | ((bits << 4) & 0xF0) | LCD_BACKLIGHT

    bus.write_byte(I2C_ADDR, bits_high)
    lcd_toggle_enable(bits_high)

    bus.write_byte(I2C_ADDR, bits_low)
    lcd_toggle_enable(bits_low)


def lcd_toggle_enable(bits):
    time.sleep(E_DELAY)
    bus.write_byte(I2C_ADDR, bits | ENABLE)
    time.sleep(E_PULSE)
    bus.write_byte(I2C_ADDR, bits & ~ENABLE)
    time.sleep(E_DELAY)


def lcd_string(message, line):
    message = message.ljust(LCD_WIDTH, " ")
    lcd_byte(line, LCD_CMD)

    for i in range(LCD_WIDTH):
        lcd_byte(ord(message[i]), LCD_CHR)


def main():
    lcd_init()

    while True:
        lcd_string("RPiSpy <", LCD_LINE_1)
        lcd_string("I2C LCD <", LCD_LINE_2)
        time.sleep(3)

        lcd_string("> RPiSpy", LCD_LINE_1)
        lcd_string("> I2C LCD", LCD_LINE_2)
        time.sleep(3)


if __name__ == "__main__":
    try:
        main()
    except KeyboardInterrupt:
        lcd_byte(0x01, LCD_CMD)
        print("Program stopped")
    finally:
        bus.close()

#web_home_automation.py#

from flask import Flask
import RPi.GPIO as GPIO

app = Flask(__name__)

RELAY_PIN = 37

GPIO.setmode(GPIO.BOARD)
GPIO.setup(RELAY_PIN, GPIO.OUT)

# Common active-low relay module
GPIO.output(RELAY_PIN, GPIO.HIGH)

HTML = """
<!DOCTYPE html>
<html>
<head>
    <title>Home Automation</title>
</head>
<body>
    <h2>Raspberry Pi Home Automation</h2>
    <p><a href="/on"><button>Turn ON</button></a></p>
    <p><a href="/off"><button>Turn OFF</button></a></p>
</body>
</html>
"""

@app.route("/")
def home():
    return HTML

@app.route("/on")
def turn_on():
    GPIO.output(RELAY_PIN, GPIO.LOW)
    return HTML + "<p>Relay ON</p>"

@app.route("/off")
def turn_off():
    GPIO.output(RELAY_PIN, GPIO.HIGH)
    return HTML + "<p>Relay OFF</p>"

try:
    app.run(host="0.0.0.0", port=5000)

finally:
    GPIO.cleanup()

#rfid_reader.py#
import subprocess
import time

def nfc_raw():
    lines = subprocess.check_output(
        "/usr/bin/nfc-poll",
        stderr=open('/dev/null', 'w')
    )
    return lines

def read_nfc():
    lines = nfc_raw()
    return lines

try:
    while True:
        myLines = read_nfc()
        buffer = []

        for line in myLines.splitlines():
            line_content = line.split()

            if not line_content[0] == b'UID':
                pass
            else:
                buffer.append(line_content)

        uid_data = buffer[0]
        id_str = (
            uid_data[2].decode() +
            uid_data[3].decode() +
            uid_data[4].decode() +
            uid_data[5].decode()
        )
        print(id_str)

except KeyboardInterrupt:
    pass
